\subsubsection*{Multiplication by a scalar}
If $y = ku(x)$ where $k$ is a scalar then
$$\dfrac{dy}{dx} = k\dfrac{du}{dx}$$

\subsubsection*{Product rule}
Given $y = u(x)v(x)$ then
$$\dfrac{dy}{dx} = u\dfrac{dv}{dx} + v\dfrac{du}{dx} $$

\subsubsection*{Quotient rule}
Given $y=\dfrac{u(x)}{v(x)}$ then
$$\dfrac{dy}{dx} = \dfrac{v\dfrac{du}{dx} - u\dfrac{dv}{dx}}{v^2}$$

\subsubsection*{Chain rule}
Given that $y$ is a function of $u$ and $u$ is a function of $x$ then
$$\dfrac{dy}{dx} = \dfrac{dy}{du} \dfrac{du}{dx}$$

\subsubsection*{Sum/difference rules}
$$\frac{d}{dx}(u \pm v) = \frac{du}{dx} \pm \frac{dv}{dx}$$

\subsubsection*{Partial differentiation}
If $z = f(x,y)$ then
$$\delta z \approx \frac{\partial f}{\partial x}\delta x + \frac{\partial f}{\partial y} \delta y$$
If $x$ and $y$ are functions of $t$ then
$$\frac{df}{dt} = \frac{\partial f}{\partial x}\frac{dx}{dt} + \frac{\partial f}{\partial y}\frac{dy}{dt}$$

\subsubsection*{L'Hopital's rule}
If $f(a)=g(a)=0$ then
$$\lim_{x \rightarrow a} \frac{f(x)}{g(x)} = \frac{f'(a)}{g'(a)}$$

\subsubsection*{Integration}
\begin{align*}
	\int [f(x)]^n f'(x) \, dx &= \left\{ \begin{array}{ll} \dfrac{[f(x)]^{n+1}}{n+1} + \mbox{constant}		& \qquad n \neq 1 \\ \log_e [f(x)] + \mbox{constant} & \qquad n = -1 \end{array} \right.
\end{align*}

\subsubsection*{Integration by parts}
If $u$ and $v$ are functions of $x$ then
$$\int u \frac{dv}{dx} = uv - \int v \frac{du}{dx} \, dx$$

\subsubsection*{Definite integration}
$$\int_{a}^{b} f(x) \, dx = \left[ \int f(x) \, dx \right]_a^b \qquad \mbox{where} \qquad
\left[f(x)\right]_a^b = f(b) - f(a)$$

\subsubsection*{Area under a curve}
The area under the curve $y=f(x)$ between $x=x_1$ and $x=x_2$ is
$$\mbox{area under curve}=\int_{x_1}^{x_2} f(x) \, dx$$
where $f(x) \geq 0$ for all $x \in [x_1,x_2]$

\subsubsection*{Volume of revolution}
The volume of revolution between the function $y=f(x)$ and the $x$-axis between $x=x_1$ and $x=x_2$ is
$$\mbox{volume of revolution}=\int_{x_1}^{x_2} \pi [f(x)]^2 \, dx$$
The volume of revolution between the function $x=f(y)$ and the $y$-axis between $y=y_1$ and $y=y_2$ is
$$\mbox{volume of revolution}=\int_{y_1}^{y_2} \pi [f(y)]^2 \, dy$$

\subsubsection*{Mean value of a function}
The mean value of the function $y=f(x)$ between $x=x_1$ and $x=x_2$ is
$$\mbox{mean value}=\frac{1}{x_2 - x_1}\int_{x_1}^{x_2}f(x) \, dx$$

\subsubsection*{Length of arc}
The length of an arc defined by the function $y=f(x)$ between $x=x_1$ and $x=x_2$ is
$$\mbox{length of arc} = \int_{x_1}^{x_2}\sqrt{1+ \left( \frac{dy}{dx} \right)^2}\, dx = \int_{t_1}^{t_2} \sqrt{\left(\frac{dx}{dt}\right)^2 + \left( \frac{dy}{dt} \right)^2} \, dt$$

\subsubsection*{Centroid of an area $(\bar{x},\bar{y})$}
$$\bar{x}=\frac{\displaystyle\int xy \, dx}{\displaystyle\int y \, dx}, \qquad \bar{y} = \frac{\dfrac{1}{2}\displaystyle\int y^2 \, dx}{\displaystyle\int y \, dx}$$

\subsubsection*{Root mean square}
Root mean square value of the function $y=f(x)$ between $x=x_1$ and $x=x_2$
$$f(x)_{rms} = \sqrt{\frac{1}{x_2-x_1}\int_{x_1}^{x_2}[f(x)]^2 \, dx}$$
